{{- /* Responsive image from assets/images. Params: src (path under assets/images), alt, class, sizes, lazy (default true), widths */ -}} {{- $img := resources.Get (printf "images/%s" .src) -}} {{- if not $img }}{{ errorf "img partial: assets/images/%s not found" .src }}{{ end -}} {{- $widths := .widths | default (slice 480 800 1200 1680) -}} {{- $srcs := slice -}} {{- range $widths -}} {{- if le . $img.Width -}} {{- $r := $img.Resize (printf "%dx webp q82" .) -}} {{- $srcs = $srcs | append (printf "%s %dw" $r.RelPermalink .) -}} {{- end -}} {{- end -}} {{- $fbw := cond (gt $img.Width 1200) 1200 $img.Width -}} {{- $fb := $img.Resize (printf "%dx webp q82" $fbw) -}}
